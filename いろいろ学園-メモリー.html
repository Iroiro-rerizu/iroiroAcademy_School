function doPost(e) { try { var params = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var type = params.type; if (type === "sync") { var sheet = getOrCreateSheet(ss, params.id); // 列: 0日時, 1ID, 2名前, 3パス, 4Lv, 5金, 6場所, 7アイコン, 8フレンド, 9届申請, 10送申請, 11X, 12Z, 13部屋ID sheet.appendRow([new Date(), params.id, params.name, params.pass, params.level, params.coins, params.location, params.avatar, params.friends, params.pending, params.sentRequests, params.x, params.z, params.roomId]); return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON); } if (type === "request") { var targetSheet = getOrCreateSheet(ss, params.targetId); var mySheet = getOrCreateSheet(ss, params.myId); updateList(targetSheet, 9, params.myId, true); // 相手の「届申請」へ updateList(mySheet, 10, params.targetId, true); // 自分の「送申請」へ return ContentService.createTextOutput(JSON.stringify({result: "requested"})); } if (type === "accept") { var mySheet = getOrCreateSheet(ss, params.myId); var targetSheet = getOrCreateSheet(ss, params.targetId); updateList(mySheet, 9, params.targetId, false); // 申請消去 updateList(mySheet, 8, params.targetId, true); // フレンド追加 updateList(targetSheet, 10, params.myId, false); // 相手の送申請消去 updateList(targetSheet, 8, params.myId, true); // 相手に自分を追加 return ContentService.createTextOutput(JSON.stringify({result: "accepted"})); } } catch(f) { return ContentService.createTextOutput(JSON.stringify({result: "error"})); } } function updateList(sheet, colIndex, value, isAdd) { var lastRow = sheet.getLastRow(); var data = (lastRow <= 1) ? [new Date(), sheet.getName(), "ユーザー", "", 1, 0, "Home", "", "", "", "", 0, 0, ""] : sheet.getRange(lastRow, 1, 1, 14).getValues()[0]; var list = data[colIndex] ? String(data[colIndex]).split(',') : []; list = list.filter(function(x){ return x !== "" && x !== "undefined"; }); if (isAdd) { if (list.indexOf(value) === -1) list.push(value); } else { list = list.filter(function(item) { return item !== value; }); } data[0] = new Date(); data[colIndex] = list.join(','); sheet.appendRow(data); } function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = e.parameter.action; if (action === "getList") { var gData = ss.getSheetByName("ゲーム選択").getDataRange().getValues(); return ContentService.createTextOutput(JSON.stringify(gData.slice(1).map((r, i) => ({id: i+2, title: r[0], desc: r[2]})))).setMimeType(ContentService.MimeType.JSON); } if (action === "getCode") { return ContentService.createTextOutput(JSON.stringify({code: ss.getSheetByName("ゲーム選択").getRange(e.parameter.row, 2).getValue()})).setMimeType(ContentService.MimeType.JSON); } if (e.parameter.roomId) { var results = []; var sheets = ss.getSheets(); for (var i = 0; i < sheets.length; i++) { var s = sheets[i]; if (s.getLastRow() <= 1) continue; var d = s.getRange(s.getLastRow(), 1, 1, 14).getValues()[0]; if (String(d[13]) === e.parameter.roomId && (new Date() - new Date(d[0])) < 60000) { results.push({id: d[1], name: d[2], x: d[11], z: d[12], location: d[6], avatar: d[7]}); } } return ContentService.createTextOutput(JSON.stringify({result: "success", members: results})); } var id = e.parameter.id; var sheet = ss.getSheetByName(id); if (!sheet || sheet.getLastRow() <= 1) return ContentService.createTextOutput(JSON.stringify({result: "not_found"})); var d = sheet.getRange(sheet.getLastRow(), 1, 1, 14).getValues()[0]; return ContentService.createTextOutput(JSON.stringify({result: "success", name: d[2], pass: d[3], level: d[4], location: d[6], avatar: d[7], friends: d[8], pending: d[9], sentRequests: d[10], lastSeen: d[0]})); } function getOrCreateSheet(ss, id) { var s = ss.getSheetByName(id); if (!s) { s = ss.insertSheet(id); s.appendRow(["日時", "ID", "名前", "パス", "Lv", "金", "場所", "アイコン", "フレンド", "届申請", "送申請", "X", "Z", "部屋ID"]); } return s; }
